        /* ==========================================================================
           2. INITIALIZATION & BROADCAST LISTENERS
           ========================================================================== */
        window.onload = function() {
            // Dengarkan update realtime dari tab lain

        function switchRole(role) {
            const studentBtn = document.getElementById('roleStudentBtn');
            const adminBtn = document.getElementById('roleAdminBtn');
            const studentForm = document.getElementById('studentLoginForm');
            const adminForm = document.getElementById('adminLoginForm');

            if (role === 'student') {

                adminForm.classList.add('hidden');

                adminBtn.className = "py-2.5 rounded-lg font-semibold text-sm transition-all bg-indigo-600 text-white shadow-md";
                studentBtn.className = "py-2.5 rounded-lg font-semibold text-sm transition-all text-slate-400 hover:text-white";
                adminForm.classList.remove('hidden');
                studentForm.classList.add('hidden');
            }
        }

        function handleStudentLogin(e) {
            e.preventDefault();
            const name = document.getElementById('studentName').value.trim();
            const nis = document.getElementById('studentNis').value.trim();

            if (!name || !nis) return;

            currentUser = {
                id: 'std_' + nis,
                name: name,
                nis: nis,
                role: 'student'
            };

            // Simpan / Inisialisasi Data Siswa di LocalStorage
            let allStudents = getStoredStudents();
            if (!allStudents[currentUser.id]) {
                allStudents[currentUser.id] = {
                    id: currentUser.id,
                    name: currentUser.name,
                    nis: currentUser.nis,
                    status: 'Sedang Mengerjakan',
                    answers: {},
                    score: 0,
                    violations: 0,
                    startTime: new Date().toLocaleTimeString()
                };
                saveStudentsData(allStudents);
            } else {
                // Load progres jika ada
                const existing = allStudents[currentUser.id];
                studentAnswers = existing.answers || {};
                violationCount = existing.violations || 0;
            }

            document.getElementById('userInfoHeader').classList.remove('hidden');
            document.getElementById('userNameHeader').innerText = `${name} (${nis})`;
            document.getElementById('loginSection').classList.add('hidden');
            document.getElementById('studentExamSection').classList.remove('hidden');

            // Aktifkan Ujian & Proteksi Security
            startExam();
        }

            if (pin !== '12345') {
                alert('Kode PIN Admin Salah! (Default: 12345)');
                return;
            }

            currentUser = { name: 'Guru Pengawas', role: 'admin' };
            document.getElementById('userInfoHeader').classList.remove('hidden');
            document.getElementById('userNameHeader').innerText = 'Administrator Ujian';
            document.getElementById('loginSection').classList.add('hidden');
            document.getElementById('adminDashboardSection').classList.remove('hidden');

        function logout() {
            if (isExamActive) {
                if (!confirm("Ujian sedang berlangsung! Yakin ingin keluar? Activitas akan dihentikan.")) return;
            }
            window.location.reload();
        }

        /* ==========================================================================
           4. EXAM ENGINE & SECURITY (ANTI-CHEAT)
           ========================================================================== */
        function startExam() {
            isExamActive = true;
            document.getElementById('violationCountBadge').innerText = `${violationCount} Kali`;

            // Render Soal Pertama
            renderQuestion();

            // Jalankan Timer
            startExamTimer();

            // Aktifkan Pengawasan Anti-Cheat Events
            initAntiCheatListeners();

            // Minta Fullscreen otomatis
            requestExamFullscreen();

            notifyDataChange();
        }

        function renderQuestion() {
            const q = EXAM_QUESTIONS[currentQuestionIndex];
            document.getElementById('questionProgress').innerText = `Soal ${currentQuestionIndex + 1} dari ${EXAM_QUESTIONS.length}`;
            document.getElementById('questionText').innerText = q.question;

            const optionsContainer = document.getElementById('optionsContainer');
            optionsContainer.innerHTML = '';

            q.options.forEach((opt, idx) => {
                const isSelected = studentAnswers[currentQuestionIndex] === idx;
                const optBtn = document.createElement('button');
                optBtn.className = `w-full text-left p-4 rounded-xl border transition-all flex items-center justify-between text-sm ${
                    isSelected 
                        ? 'bg-indigo-600/20 border-indigo-500 text-indigo-200 font-semibold' 
                        : 'bg-slate-900/60 border-slate-700/80 text-slate-300 hover:border-slate-500'
                }`;
                optBtn.onclick = () => selectAnswer(idx);

                optBtn.innerHTML = `
                    <div class="flex items-center gap-3">
                        <span class="w-7 h-7 rounded-lg border text-xs flex items-center justify-center ${
                            isSelected ? 'bg-indigo-600 border-indigo-500 text-white' : 'border-slate-600 text-slate-400'
                        }">${String.fromCharCode(65 + idx)}</span>
                        <span>${opt}</span>
                    </div>
                    ${isSelected ? '<i class="fa-solid fa-circle-check text-indigo-400"></i>' : ''}
                `;
                optionsContainer.appendChild(optBtn);
            });

            // Handle Tombol Navigasi
            document.getElementById('prevBtn').disabled = currentQuestionIndex === 0;
            if (currentQuestionIndex === EXAM_QUESTIONS.length - 1) {
                document.getElementById('nextBtn').classList.add('hidden');
                document.getElementById('submitExamBtn').classList.remove('hidden');
            } else {
                document.getElementById('nextBtn').classList.remove('hidden');
                document.getElementById('submitExamBtn').classList.add('hidden');
            }
        }

        function selectAnswer(optionIndex) {
            if (!isExamActive) return;
            studentAnswers[currentQuestionIndex] = optionIndex;
            
            // Simpan State Ke Storage
            updateCurrentStudentState({ answers: studentAnswers });
            renderQuestion();
        }

        function navigateQuestion(direction) {
            currentQuestionIndex += direction;
            renderQuestion();
        }

        function startExamTimer() {
            examTimerInterval = setInterval(() => {
                timeLeft--;
                const mins = Math.floor(timeLeft / 60);
                const secs = timeLeft % 60;
                document.getElementById('examTimer').innerText = 
                    `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;

                if (timeLeft <= 0) {
                    clearInterval(examTimerInterval);
                    autoSubmitExam();
                }
            }, 1000);
        }

        /* --------------------------------------------------------------------------
           ANTI-CHEAT ENGINE (BLUR, VISIBILITY, FULLSCREEN, KEYBOARD)
           -------------------------------------------------------------------------- */
        function initAntiCheatListeners() {
            // 1. Deteksi Pindah Tab (Visibility Change)
            document.addEventListener('visibilitychange', () => {
                if (isExamActive && document.hidden) {
                    registerViolation("Membuka Tab Lain / Minimalkan Browser");
                }
            });

            // 2. Deteksi Unfocus Window (Klik Diluar / Alt+Tab)
            window.addEventListener('blur', () => {
                if (isExamActive) {
                    registerViolation("Berpindah Jendela Aplikasi (Blur Event)");
                }
            });

            // 3. Deteksi Keluar Fullscreen
            document.addEventListener('fullscreenchange', () => {
                if (isExamActive && !document.fullscreenElement) {
                    registerViolation("Keluar dari Layar Fullscreen Ujian");
                }
            });

            // 4. Blokir Klik Kanan (Context Menu)
            document.addEventListener('contextmenu', (e) => {
                if (isExamActive) {
                    e.preventDefault();
                    return false;
                }
            });

            // 5. Blokir Copy Paste
            document.addEventListener('copy', (e) => isExamActive && e.preventDefault());
            document.addEventListener('paste', (e) => isExamActive && e.preventDefault());
        }

        function initKeyboardProtections() {
            // Blokir Shortcut Penting: F12, Ctrl+Shift+I, Ctrl+Tab, Ctrl+N, Ctrl+T, Alt+Tab
            window.addEventListener('keydown', (e) => {
                if (!isExamActive) return;

                // F12 atau DevTools
                if (e.key === 'F12' || (e.ctrlKey && e.shiftKey && (e.key === 'I' || e.key === 'i' || e.key === 'J' || e.key === 'j'))) {
                    e.preventDefault();
                    registerViolation("Mencoba Membuka DevTools (Developer Tools)");
                }

                // Shortcut Pindah / Buka Tab Baru
                if (e.ctrlKey && (e.key === 't' || e.key === 'T' || e.key === 'n' || e.key === 'N' || e.key === 'w' || e.key === 'W')) {
                    e.preventDefault();
                    registerViolation("Shortcut Tab Browser Diblokir");
                }
            });
        }

        function registerViolation(reason) {
            if (!isExamActive) return;
            violationCount++;
            
            // Update Tampilan UI Siswa
            document.getElementById('violationCountBadge').innerText = `${violationCount} Kali`;
            document.getElementById('cheatAlertModal').classList.remove('hidden');

            // Log & Sync Ke LocalStorage & Admin
            updateCurrentStudentState({ 
                violations: violationCount,
                lastViolationReason: reason,
                lastViolationTime: new Date().toLocaleTimeString()
            });

            logViolationEvent(`${currentUser.name} (${currentUser.nis}) melanggar: ${reason}`);
        }

        function closeCheatModal() {
            document.getElementById('cheatAlertModal').classList.add('hidden');
            requestExamFullscreen();
        }

        function requestExamFullscreen() {
            const elem = document.documentElement;
            if (elem.requestFullscreen) {
                elem.requestFullscreen().catch(err => console.log("Fullscreen blocked"));
            }
        }

        /* --------------------------------------------------------------------------
           SUBMIT UJIAN
           -------------------------------------------------------------------------- */
        function confirmSubmitExam() {
            if (confirm("Apakah Anda yakin ingin menyelesaikan ujian sekarang?")) {
                finishExam();
            }
        }

        function autoSubmitExam() {
            alert("Waktu ujian telah habis! Sistem secara otomatis menyimpan jawaban Anda.");
            finishExam();
        }

        function finishExam() {
            isExamActive = false;
            clearInterval(examTimerInterval);

            // Hitung Score Akhir
            let score = 0;
            EXAM_QUESTIONS.forEach((q, idx) => {
                if (studentAnswers[idx] === q.answer) {
                    score += (100 / EXAM_QUESTIONS.length);
                }
            });

            // Update State Akhir Siswa
            updateCurrentStudentState({
                status: 'Selesai',
                score: score,
                violations: violationCount
            });

            // Tampilkan Screen Hasil
            document.getElementById('studentExamSection').classList.add('hidden');
            document.getElementById('studentResultSection').classList.remove('hidden');
            document.getElementById('finalScoreText').innerText = score;
            document.getElementById('finalViolationsText').innerText = violationCount;

            // Exit Fullscreen
            if (document.exitFullscreen) {
                document.exitFullscreen().catch(e => {});
            }
        }

        /* ==========================================================================
           5. ADMIN DASHBOARD REALTIME ENGINE
           ========================================================================== */
        function renderAdminDashboard() {
            const students = getStoredStudents();
            const tableBody = document.getElementById('adminTableBody');
            tableBody.innerHTML = '';

            let studentList = Object.values(students);
            let totalStudents = studentList.length;
            let activeCount = 0;
            let completedCount = 0;
            let totalViolations = 0;

            if (totalStudents === 0) {
                tableBody.innerHTML = `
                    <tr>
                        <td colspan="7" class="text-center py-8 text-slate-500">
                            Belum ada siswa yang bergabung / melakukan ujian.
                        </td>
                    </tr>`;
            } else {
                studentList.forEach(std => {
                    if (std.status === 'Sedang Mengerjakan') activeCount++;
                    if (std.status === 'Selesai') completedCount++;
                    totalViolations += (std.violations || 0);

                    const answeredCount = Object.keys(std.answers || {}).length;
                    const progressPercent = Math.round((answeredCount / EXAM_QUESTIONS.length) * 100);

                    const row = document.createElement('tr');
                    row.className = "hover:bg-slate-700/30 transition-all";

                    row.innerHTML = `
                        <td class="py-4 px-6 font-semibold text-white flex items-center gap-2">
                            <i class="fa-solid fa-user-graduate text-slate-400"></i> ${std.name}
                        </td>
                        <td class="py-4 px-6 font-mono text-xs text-slate-400">${std.nis}</td>
                        <td class="py-4 px-6">
                            ${std.status === 'Selesai' 
                                ? '<span class="bg-emerald-500/10 text-emerald-400 border border-emerald-500/30 px-2.5 py-1 rounded-full text-xs font-semibold">Selesai</span>'
                                : '<span class="bg-amber-500/10 text-amber-400 border border-amber-500/30 px-2.5 py-1 rounded-full text-xs font-semibold animate-pulse">Mengerjakan</span>'
                            }
                        </td>
                        <td class="py-4 px-6">
                            <div class="w-full bg-slate-700 rounded-full h-2 max-w-[100px] mb-1">
                                <div class="bg-indigo-500 h-2 rounded-full" style="width: ${progressPercent}%"></div>
                            </div>
                            <span class="text-xs text-slate-400 font-mono">${answeredCount}/${EXAM_QUESTIONS.length} Soal</span>
                        </td>
                        <td class="py-4 px-6">
                            ${std.violations > 0 
                                ? `<span class="bg-red-500/20 text-red-400 border border-red-500/30 px-2.5 py-1 rounded-full text-xs font-bold flex items-center gap-1 w-max">
                                    <i class="fa-solid fa-triangle-exclamation"></i> ${std.violations} Violations
                                   </span>`
                                : '<span class="text-xs text-slate-500">Bersih (0)</span>'
                            }
                        </td>
                        <td class="py-4 px-6 font-bold font-mono text-indigo-400 text-base">
                            ${std.status === 'Selesai' ? std.score : '-'}
                        </td>
                        <td class="py-4 px-6 text-center">
                            <button onclick="resetStudentExam('${std.id}')" class="bg-slate-700 hover:bg-red-600/80 text-slate-300 hover:text-white px-3 py-1.5 rounded-lg text-xs transition-all flex items-center gap-1 mx-auto">
                                <i class="fa-solid fa-rotate-left"></i> Reset
                            </button>
                        </td>
                    `;
                    tableBody.appendChild(row);
                });
            }

            // Update Counter Stats Header
            document.getElementById('statTotalStudents').innerText = totalStudents;
            document.getElementById('statActiveStudents').innerText = activeCount;
            document.getElementById('statCompletedStudents').innerText = completedCount;
            document.getElementById('statTotalViolations').innerText = totalViolations;
        }

        function resetStudentExam(studentId) {
            if (confirm("Riset akses ujian siswa ini? Data pengerjaan akan dihapus dan siswa bisa mengulang dari awal.")) {
                let students = getStoredStudents();
                delete students[studentId];
                saveStudentsData(students);
                renderAdminDashboard();
            }
        }

        function clearAllData() {
            if (confirm("PERINGATAN: Apakah Anda yakin ingin menghapus SELURUH data ujian siswa?")) {
                localStorage.removeItem('uas_students_db');
                localStorage.removeItem('uas_violation_logs');
                renderAdminDashboard();
                document.getElementById('violationLogBox').innerHTML = '<div class="text-slate-500">[System Status] Data telah dibersihkan.</div>';
            }
        }

        function exportToCSV() {
            const students = Object.values(getStoredStudents());
            if (students.length === 0) {
                alert("Tidak ada data siswa untuk diekspor!");
                return;
            }

            let csvContent = "data:text/csv;charset=utf-8,";
            csvContent += "NIS,Nama Siswa,Status,Nilai,Jumlah Pelanggaran\n";

            students.forEach(s => {
                csvContent += `"${s.nis}","${s.name}","${s.status}","${s.score || 0}","${s.violations || 0}"\n`;
            });

            const encodedUri = encodeURI(csvContent);
            const link = document.createElement("a");
            link.setAttribute("href", encodedUri);
            link.setAttribute("download", `Rekap_UAS_${new Date().toISOString().slice(0,10)}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        /* ==========================================================================
           6. LOCALSTORAGE & UTILITY HELPERS
           ========================================================================== */
        function getStoredStudents() {
            return JSON.parse(localStorage.getItem('uas_students_db') || '{}');
        }

        function saveStudentsData(data) {
            localStorage.setItem('uas_students_db', JSON.stringify(data));
            notifyDataChange();
        }

        function updateCurrentStudentState(partialData) {
            if (!currentUser || currentUser.role !== 'student') return;
            let students = getStoredStudents();
            if (students[currentUser.id]) {
                students[currentUser.id] = { ...students[currentUser.id], ...partialData };
                saveStudentsData(students);
            }
        }

        function notifyDataChange() {
            // Broadcast event ke tab lain (Dashboard Guru)
            syncChannel.postMessage({ type: 'SYNC_DATA' });
        }

        function logViolationEvent(msg) {
            const logBox = document.getElementById('violationLogBox');
            const time = new Date().toLocaleTimeString();
            const logItem = document.createElement('div');
            logItem.className = "text-red-400 bg-red-500/10 p-1.5 rounded border border-red-500/20";
            logItem.innerHTML = `[${time}] ⚠️ ${msg}`;
            
            if (logBox) {
                logBox.prepend(logItem);
            }
        }

        function toggleShareModal() {
            const modal = document.getElementById('shareModal');
            modal.classList.toggle('hidden');
        }
